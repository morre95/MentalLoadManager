import { useQuery } from "@tanstack/react-query";
import KanbanBoard from "@/components/dashboard/KanbanBoard";

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL; // recommended

async function fetchTasks(token) {
  const res = await fetch(`${API_BASE_URL}/tasks`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!res.ok) throw new Error("Failed to fetch tasks");
  return res.json();
}

export default function Dashboard() {
  const token = localStorage.getItem("token"); // or however you store auth
  const isLoggedIn = Boolean(token);

  const { data } = useQuery({
    queryKey: ["tasks"],
    queryFn: () => fetchTasks(token),
    enabled: isLoggedIn,
  });

  // You might need to transform API format -> {todo, doing, done}
  const tasksByStatus = data
    ? {
        todo: data.filter((t) => t.status === "todo"),
        doing: data.filter((t) => t.status === "doing"),
        done: data.filter((t) => t.status === "done"),
      }
    : undefined;

  return <KanbanBoard tasks={tasksByStatus} />;
}
